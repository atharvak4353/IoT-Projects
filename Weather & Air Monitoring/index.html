About the Project

This project demonstrates an IoT-based Weather & Air Monitoring System designed to collect environmental data using multiple sensors.

The Arduino UNO R4 WiFi acts as the main embedded controller. A DHT11 sensor measures temperature and humidity, while MQ135, MQ2 and MQ9 sensors provide analog readings for air-quality and gas monitoring.

The collected readings are processed by the Arduino and transmitted over Wi-Fi to the Blynk IoT platform. This allows the sensor values to be viewed remotely from a dashboard.

The project combines embedded programming, sensor interfacing, wireless communication and IoT dashboard integration in a single practical system.

How It Works

1. Sensor Acquisition

Arduino reads temperature, humidity and analog MQ sensor values.

2. Data Processing

Raw analog values are mapped into a 0–100 monitoring range.

3. Wi-Fi Connection

Arduino UNO R4 WiFi connects to the configured wireless network.

4. Cloud Upload

Processed readings are sent to Blynk virtual pins.

5. Dashboard Display

Blynk displays the values using individual gauge widgets.

6. Continuous Update

The process repeats automatically every two seconds.

Arduino Firmware

weather_air_monitoring.ino 
#define BLYNK_PRINT Serial

#define BLYNK_TEMPLATE_ID "YOUR_TEMPLATE_ID"
#define BLYNK_TEMPLATE_NAME "Weather & Air Monitoring"
#define BLYNK_AUTH_TOKEN "YOUR_BLYNK_AUTH_TOKEN"

#include <SPI.h>
#include <WiFiS3.h>
#include <BlynkSimpleWifi.h>
#include <DHT.h>

char ssid[] = "YOUR_WIFI_NAME";
char pass[] = "YOUR_WIFI_PASSWORD";

#define DHTPIN 4
#define DHTTYPE DHT11

DHT dht(
    DHTPIN,
    DHTTYPE
);

/*
    Sensor Connections

    MQ135 -> A0
    MQ2   -> A2
    MQ9   -> A3
*/

const int mq135 = A0;
const int mq2   = A2;
const int mq9   = A3;

BlynkTimer timer;


void sendSensorData()
{

    float mq135Value =
        analogRead(mq135);

    float mq2Value =
        analogRead(mq2);

    float mq9Value =
        analogRead(mq9);


    mq135Value = map(
        mq135Value,
        0,
        1023,
        0,
        100
    );


    mq2Value = map(
        mq2Value,
        0,
        1023,
        0,
        100
    );


    mq9Value = map(
        mq9Value,
        0,
        1023,
        0,
        100
    );


    float temperature =
        dht.readTemperature();


    float humidity =
        dht.readHumidity();


    if (
        isnan(temperature) ||
        isnan(humidity)
    )
    {

        Serial.println(
            "DHT read failed"
        );

        return;

    }


    Blynk.virtualWrite(
        V0,
        temperature
    );


    Blynk.virtualWrite(
        V1,
        humidity
    );


    Blynk.virtualWrite(
        V2,
        mq135Value
    );


    Blynk.virtualWrite(
        V3,
        mq2Value
    );


    Blynk.virtualWrite(
        V4,
        mq9Value
    );


    Serial.print("Temperature: ");
    Serial.print(temperature);

    Serial.print(" C | Humidity: ");
    Serial.print(humidity);

    Serial.print(" % | MQ135: ");
    Serial.print(mq135Value);

    Serial.print(" | MQ2: ");
    Serial.print(mq2Value);

    Serial.print(" | MQ9: ");
    Serial.println(mq9Value);

}


void setup()
{

    Serial.begin(9600);


    pinMode(
        mq135,
        INPUT
    );


    pinMode(
        mq2,
        INPUT
    );


    pinMode(
        mq9,
        INPUT
    );


    dht.begin();


    Blynk.begin(
        BLYNK_AUTH_TOKEN,
        ssid,
        pass
    );


    timer.setInterval(
        2000L,
        sendSensorData
    );

}


void loop()
{

    Blynk.run();

    timer.run();

}
